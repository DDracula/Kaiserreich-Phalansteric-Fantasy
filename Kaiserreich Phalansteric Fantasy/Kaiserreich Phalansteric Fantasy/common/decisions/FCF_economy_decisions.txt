    FCF_economy_focus_on_industry = {
		icon = ger_mefo_bills
		priority = 3
		allowed = { tag = FRA }
		available = { 
			custom_trigger_tooltip = {
				tooltip = can_choose_more_economy_focus
				check_variable = { econfocus < maxeconfocus } 
			}
		}
		ai_will_do = { base = 10 }
		custom_cost_trigger  = { NOT = { check_variable = { economy_power < 10 } } }
		custom_cost_text = cost_ep_10
		cancel_trigger = {
			custom_trigger_tooltip = {
				tooltip = stop_economy_focus
				NOT = { has_country_flag = economy_focus_on_industry }
			}
		}
		days_remove = 90
		modifier = {
			monthly_change_eib = 0.5                        
			monthly_change_eae = 0.3
			monthly_change_eq = -0.4
			monthly_change_el = 0.3
		}
		complete_effect = {
			set_country_flag = economy_focus_on_industry
			add_to_variable = { econfocus = 1 }
			custom_effect_tooltip = economy_focus_tt
			add_to_variable = { 
				economy_power = -10 
				tooltip = economy_power_tt
			}
			
			custom_effect_tooltip = effect_society
			set_temp_variable = { wc = 2 }
			set_temp_variable = { pc = -1 }
			set_temp_variable = { mc = -1 }
			set_temp_variable = { sc = 0 }
			set_temp_variable = { bc = 0 }
			set_temp_variable = { ac = 0 }
			society_pop_satisfactions_change = yes
		}
		remove_effect = {
			add_to_variable = { 
				economy_power = -10 
				tooltip = economy_power_tt
			}
			custom_effect_tooltip = effect_society
			set_temp_variable = { wc = 2 }

		}       
	}
