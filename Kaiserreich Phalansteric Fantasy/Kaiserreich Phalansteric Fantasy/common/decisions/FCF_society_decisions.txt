	FCF_society_pop_workers_mission = {
		icon = generic_civil_support
		days_mission_timeout = 180
		activation = { has_country_flag = during_pop_workers_mission }
		available = {
			custom_trigger_tooltip = {
				tooltip = FCF_society_pop_workers_mission_available_trigger
				NOT = { check_variable = { workers_mission_investment_count < workers_mission_target_investment_count } }
			}
		}
		is_good = no
		fixed_random_seed = no
		complete_effect = {
			custom_effect_tooltip = effect_society
			set_temp_variable = { pop_workers_change = { value = workers_mission_reward_level multiply = 1 } }
			set_temp_variable = { pop_peasants_change = { value = workers_mission_reward_level multiply = -0.2 } }
			set_temp_variable = { pop_merchants_change = { value = workers_mission_reward_level multiply = -0.2 } }
			set_temp_variable = { pop_scholars_change = { value = workers_mission_reward_level multiply = -0.2 } }
			set_temp_variable = { pop_bureaucrats_change = { value = workers_mission_reward_level multiply = -0.2 } }
			set_temp_variable = { pop_army_change = { value = workers_mission_reward_level multiply = -0.2 } }
			society_pop_influence_change = yes

			custom_effect_tooltip = tooltip_white_line
			set_temp_variable = { pop_satisfaction_workers_change = { value = workers_mission_reward_level multiply = 1.5 } }
			set_temp_variable = { pop_satisfaction_peasants_change = 0 }
			set_temp_variable = { pop_satisfaction_merchants_change = 0 }
			set_temp_variable = { pop_satisfaction_scholars_change = 0 }
			set_temp_variable = { pop_satisfaction_bureaucrats_change = 0 }
			set_temp_variable = { pop_satisfaction_army_change = 0 }
			society_pop_satisfaction_change = yes

			add_to_variable = { workers_mission_reward_level = 1 }
			add_to_variable = { workers_mission_target_investment_count = 3 }

			clr_country_flag = during_pop_workers_mission
		}
		timeout_effect = {
			custom_effect_tooltip = effect_society
			set_temp_variable = { pop_satisfaction_workers_change = -3 }
			set_temp_variable = { pop_satisfaction_peasants_change = 0 }
			set_temp_variable = { pop_satisfaction_merchants_change = 0 }
			set_temp_variable = { pop_satisfaction_scholars_change = 0 }
			set_temp_variable = { pop_satisfaction_bureaucrats_change = 0 }
			set_temp_variable = { pop_satisfaction_army_change = 0 }
			society_pop_satisfaction_change = yes

			set_variable = { workers_mission_reward_level = 1 }
			set_variable = { workers_mission_target_investment_count = 3 }

			clr_country_flag = during_pop_workers_mission
		}
		ai_will_do = { base = 10 }
