# 非法国国家用于回应国际干涉行动的反制决议组。

	icon = generic_political_actions
	priority = 9
	allowed = { NOT = { tag = FRA } }
	visible = {
		OR = {
			AND = {
				has_country_flag = FCF_II_HAD
				has_socialist_government = no
				NOT = { has_country_flag = ii_has_finished }
			}
			AND = {
				OR = {
					tag = IRE
					tag = ENG
				}
				IRE = { has_country_flag = ii_ireland_revolution }
			}
			AND = {
				OR = {
					tag = COR
					tag = NFA
				}
				COR = {
					is_puppet_of = NFA
					has_idea = FCF_state_of_mafia
				}
			}
		}  
	}
	visible_when_empty = yes        
}
