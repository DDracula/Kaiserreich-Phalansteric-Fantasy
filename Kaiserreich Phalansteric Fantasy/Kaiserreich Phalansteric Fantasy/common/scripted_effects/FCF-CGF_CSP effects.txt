	CGF_vs_CSP_value_change = {
		add_to_variable = { FCF_CGF_vs_CSP_value = CvCc }
		custom_effect_tooltip = CGF_vs_CSP_value_change_tips
		clamp_variable = { var = FCF_CGF_vs_CSP_value max = 100 min = -100 }

			limit = {
				OR = {
					check_variable = { FCF_CGF_vs_CSP_value > 0 }

				}
			}

			add_to_variable = { gui_cgf = t }
		}
		FCF_CGF_vs_CSP_modifier_judge = yes
	}

	cgf_seat_change = {
		set_temp_variable = { seat_array^0 = 0 }
		set_temp_variable = { seatc_array^0 = 0 }
		meta_effect = {
			text = {
				for_loop_effect = {
					start = 1
					end = 11
					value = party_var
					set_temp_variable = { seat_array^party_var = seat_[PARTY] }
					set_temp_variable = { seatc_array^party_var = seatc_[PARTY] }
				}
			}
			PARTY = "[party_var_name]"
		}
		set_temp_variable = { diff_num = 0 }
		for_each_loop = {
			array = seatc_array
			value = seatc
			index = idx
			if = {
				limit = { check_expr = { value = seatc add = seat_array^idx less_than = 0 } }
				set_temp_variable = { seatc_array^idx = { value = seat_array^idx multiply = -1 } }
				add_to_temp_variable = { diff_num = { value = seatc add = seat_array^idx } }
			}
		}
		while_loop_effect = {
			limit = { check_variable = { diff_num < 0 } }
			for_each_loop = {
				array = seatc_array
				value = seatc
				index = idx
				if = {
					limit = { check_variable = { seatc > 0 } }
					add_to_variable = { seatc_array^idx = -1 }
					add_to_variable = { diff_num = 1 }
				}
			}
		}
		meta_effect = {
			text = {
				for_loop_effect = {
					start = 1
					end = 11
					value = party_var
					set_temp_variable = { seatc_[PARTY] = seatc_array^party_var }
				}
			}
			PARTY = "[party_var_name]"
		}

		if = {
			limit = { NOT = { check_variable = { seatc_CNS = 0 } } }
			add_to_variable = { seat_CNS = seatc_CNS }
			custom_effect_tooltip = cgf_seat_change_CNS_tt
		}
		if = {
			limit = { NOT = { check_variable = { seatc_PSF = 0 } } }
			add_to_variable = { seat_PSF = seatc_PSF }
			custom_effect_tooltip = cgf_seat_change_PSF_tt
		}
		if = {
			limit = { NOT = { check_variable = { seatc_POU = 0 } } }
			add_to_variable = { seat_POU = seatc_POU }
			custom_effect_tooltip = cgf_seat_change_POU_tt
		}
		if = {
			limit = { NOT = { check_variable = { seatc_CGT = 0 } } }
			add_to_variable = { seat_CGT = seatc_CGT }
			custom_effect_tooltip = cgf_seat_change_CGT_tt
		}
		if = {
			limit = { NOT = { check_variable = { seatc_PCOP = 0 } } }
			add_to_variable = { seat_PCOP = seatc_PCOP }
			custom_effect_tooltip = cgf_seat_change_PCOP_tt
		}
		if = {
			limit = { NOT = { check_variable = { seatc_UA = 0 } } }
			add_to_variable = { seat_UA = seatc_UA }
			custom_effect_tooltip = cgf_seat_change_UA_tt
		}
