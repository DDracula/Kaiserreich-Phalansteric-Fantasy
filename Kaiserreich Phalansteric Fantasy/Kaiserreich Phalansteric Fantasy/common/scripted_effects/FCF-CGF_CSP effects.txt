		set_temp_variable = { current_seat_array^0 = 0 }
		set_temp_variable = { seat_change_array^0 = 0 }
		meta_effect = {
			text = {
				for_loop_effect = {
					start = 1
					end = 11
					value = target_party_id
					set_temp_variable = { current_seat_array^target_party_id = seat_[PARTY] }
					set_temp_variable = { seat_change_array^target_party_id = seat_change_[PARTY] }
				}
			}
			PARTY = "[party_var_name]"
		}
		set_temp_variable = { unallocated_seat_deficit = 0 }
		for_each_loop = {
			array = seat_change_array
			value = seat_change_value
			index = party_index
			if = {
				limit = { check_expr = { value = seat_change_value add = current_seat_array^party_index less_than = 0 } }
				set_temp_variable = { seat_change_array^party_index = { value = current_seat_array^party_index multiply = -1 } }
				add_to_temp_variable = { unallocated_seat_deficit = { value = seat_change_value add = current_seat_array^party_index } }
			}
		}
		while_loop_effect = {
			limit = { check_variable = { unallocated_seat_deficit < 0 } }
			for_each_loop = {
				array = seat_change_array
				value = seat_change_value
				index = party_index
				if = {
					limit = { check_variable = { seat_change_value > 0 } }
					add_to_variable = { seat_change_array^party_index = -1 }
					add_to_variable = { unallocated_seat_deficit = 1 }
				}
			}
		}
		meta_effect = {
			text = {
				for_loop_effect = {
					start = 1
					end = 11
					value = target_party_id
					set_temp_variable = { seat_change_[PARTY] = seat_change_array^target_party_id }
				}
			}
			PARTY = "[party_var_name]"
		}

		if = {
			limit = { NOT = { check_variable = { seat_change_CNS = 0 } } }
			add_to_variable = { seat_CNS = seat_change_CNS }

			add_to_variable = { seat_POU = seat_change_POU }
			custom_effect_tooltip = politics_CGF_seat_change_POU_tt
		}
		if = {

			custom_effect_tooltip = politics_CGF_seat_change_CGT_tt
		}

			text = {
				set_temp_variable = { posture_attitude_change = { value = CAE_[ID]^1 multiply = diplomacy_reform_posture_change } }
				set_temp_variable = { strategy_attitude_change = { value = CAE_[ID]^2 multiply = diplomacy_reform_strategy_change } }
				set_temp_variable = { enemy_attitude_change = { value = CAE_[ID]^3 multiply = diplomacy_reform_enemy_change } }
				set_temp_variable = { total_attitude_change = {
						value = posture_attitude_change
						add = strategy_attitude_change
						add = enemy_attitude_change
						clamp = { min = -3 max = 3 }
					} }
				set_temp_variable = { target_CSP_group = 1 }
				set_temp_variable = { politics_CSP_attitude_change = total_attitude_change }
				politics_CSP_attitude_change = yes
			}
			ID = "[?csp_member_id]"
		}
	}
