	### 作用域：国家；主要作用：修改法伦施泰尔团结度并限制其取值范围。
	international_phalanstere_unity_change = {
		if = {
			limit = { NOT = { check_variable = { international_phalanstere_unity_change = 0 } } }

			add_to_variable = { global.phalanstere_unity = international_phalanstere_unity_change }
			custom_effect_tooltip = international_phalanstere_unity_change_tt

	### 作用域：国家；主要作用：扫描各国并重建法伦施泰尔成员及意识形态数组。
	international_phalanstere_member_update = {
		clear_array = global.phalanstere_member
		clear_array = global.phalanstere_totalist
		clear_array = global.phalanstere_syndicalist
		clear_array = global.phalanstere_radical_socialist
		every_country = {
			limit = { FRA_is_phalanstere_member = yes }
			add_to_array = { array = global.phalanstere_member value = THIS.id }
			if = {
				limit = { has_government = totalist }
				add_to_array = {
					array = global.phalanstere_totalist
					value = THIS.id
				}
			}
			if = {
				limit = { has_government = syndicalist }
				add_to_array = {
					array = global.phalanstere_syndicalist
					value = THIS.id
				}
			}
			if = {
				limit = { has_government = radical_socialist }
				add_to_array = {
					array = global.phalanstere_radical_socialist
					value = THIS.id
				}
			}
		}
	}

	### 作用域：国家；主要作用：按成员数量与意识形态构成计算法伦施泰尔团结度月变化。
	international_phalanstere_unity_monthly_update = {
		international_phalanstere_member_update = yes

		set_variable = { global.phauc_monthly1 = global.phalanstere_member^num }

		if = {
			limit = { phalanstere_totalist_advantage = yes }
			set_variable = { global.phauc_monthly2 = global.phalanstere_syndicalist^num }
			add_to_variable = { global.phauc_monthly2 = global.phalanstere_radical_socialist^num }
			multiply_variable = { global.phauc_monthly2 = -0.1 }
		}
		if = {
			limit = { phalanstere_syndicalist_advantage = yes }
			set_variable = { global.phauc_monthly2 = global.phalanstere_totalist^num }
			add_to_variable = { global.phauc_monthly2 = global.phalanstere_radical_socialist^num }
			multiply_variable = { global.phauc_monthly2 = -0.1 }
		}
		if = {
			limit = { phalanstere_radical_socialist_advantage = yes }
			set_variable = { global.phauc_monthly2 = global.phalanstere_totalist^num }
			add_to_variable = { global.phauc_monthly2 = global.phalanstere_radical_socialist^num }
			multiply_variable = { global.phauc_monthly2 = -0.1 }
		}
		if = {
			limit = {
				phalanstere_totalist_advantage = no
				phalanstere_syndicalist_advantage = no
				phalanstere_radical_socialist_advantage = no
			}
			set_variable = { global.phauc_monthly2 = global.phalanstere_member^num }
			multiply_variable = { global.phauc_monthly2 = -0.1 }
		}
		multiply_variable = { global.phauc_monthly2 = 5 }

		set_variable = { global.phauc_monthly = global.phauc_monthly1 }
		add_to_variable = { global.phauc_monthly = global.phauc_monthly2 }
	}

	### 作用域：国家；主要作用：汇总成员 MIO 投资并更新法伦施泰尔产业投资进度。
	international_phalanstere_industry_investment_update = {
		set_temp_variable = { global.phalanstere_industry_investment_progress_change = 0 }
		every_country = {
			limit = {
				mio:FRA_phalanstere_research_organization = {
					OR = {
						is_mio_trait_completed = mio_phalanstere_industry_investment_factory
						is_mio_trait_completed = mio_phalanstere_industry_investment_construction
					}
				}
			}
			if = {
				limit = { mio:FRA_phalanstere_research_organization = { is_mio_trait_completed = mio_phalanstere_industry_investment_factory } }
				add_to_temp_variable = { global.phalanstere_industry_investment_progress_change = 10 }
			}
			if = {
				limit = { mio:FRA_phalanstere_research_organization = { is_mio_trait_completed = mio_phalanstere_industry_investment_construction } }
				add_to_temp_variable = { global.phalanstere_industry_investment_progress_change = 5 }
			}
		}
		add_to_variable = { global.phalanstere_industry_investment_progress = global.phalanstere_industry_investment_progress_change }
		while_loop_effect = {
			limit = { check_variable = { global.phalanstere_industry_investment_progress > 99.9 } }
			add_to_variable = { global.phalanstere_industry_investment_count = 1 }
			add_to_variable = { global.phalanstere_industry_investment_progress = -100 }
		}
	}

		subtract_from_temp_variable = { other_popularity = democratic_popularity }

					has_government = social_liberal
				}

			set_temp_variable = { democratic_popularity_weight = -1 }
			set_temp_variable = { other_popularity_weight = -0.4 }

			set_temp_variable = { other_popularity_weight = -0.4 }

		multiply_temp_variable = { democratic_popularity = democratic_popularity_weight }
		multiply_temp_variable = { other_popularity = other_popularity_weight }
