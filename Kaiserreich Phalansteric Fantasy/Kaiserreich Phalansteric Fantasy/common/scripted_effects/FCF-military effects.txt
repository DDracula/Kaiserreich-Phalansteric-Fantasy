    improvement_project_progress_judge = {
        if = {
            limit = { has_war = yes }
            set_temp_variable = { FCF_caused_casualties = 0 }
            every_enemy_country = {
                if = {
                    limit = { casualties_inflicted_by = { opponent = FRA thousands > 100 } }
                    add_to_temp_variable = { FRA.FCF_caused_casualties = 6 }
                }
                else_if = {
                    limit = { casualties_inflicted_by = { opponent = FRA thousands > 80 } }
                    add_to_temp_variable = { FRA.FCF_caused_casualties = 5 }
                }
                else_if = {
                    limit = { casualties_inflicted_by = { opponent = FRA thousands > 60 } }
                    add_to_temp_variable = { FRA.FCF_caused_casualties = 4 }
                }
                else_if = {
                    limit = { casualties_inflicted_by = { opponent = FRA thousands > 40 } }
                    add_to_temp_variable = { FRA.FCF_caused_casualties = 3 }
                }
                else_if = {
                    limit = { casualties_inflicted_by = { opponent = FRA thousands > 20 } }
                    add_to_temp_variable = { FRA.FCF_caused_casualties = 2 }
                }
                else = { add_to_temp_variable = { FRA.FCF_caused_casualties = 1 } }
            }
            add_to_variable = { improving_project_progress = { value = FCF_caused_casualties multiply = 0.2 } }
        }
        else = { add_to_variable = { improving_project_progress = { value = num_of_volunteers multiply = 0.5 } } }
    }
