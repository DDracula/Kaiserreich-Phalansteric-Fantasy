                start = 1
                end = rnum
                random_list = {
                    1 = { 
                        modifier = { 
                            factor = 0 
                            NOT = { check_variable = { wc = 0 } }
                        }
                        set_temp_variable = { wc = rchange } 
                    }
                    1 = { 
                        modifier = { 
                            factor = 0
