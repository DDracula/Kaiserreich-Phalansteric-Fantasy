random_pop_satisfation_change = {
    custom_effect_tooltip = random_pop_satisfation_change_tt
    hidden_effect = {
        set_temp_variable = { wc = 0 }
        set_temp_variable = { pc = 0 }
        set_temp_variable = { mc = 0 }
        set_temp_variable = { sc = 0 }
        set_temp_variable = { bc = 0 }
        set_temp_variable = { ac = 0 }
        for_loop_effect = {
            start = 1
            end = rnum
            random_list = {
                1 = { 
                    modifier = { 
                        factor = 0 
                        NOT = { check_variable = { wc = 0 } }
                    }
                    set_temp_variable = { wc = rchange } 
                }
                1 = { 
                    modifier = { 
                        factor = 0 
                        NOT = { check_variable = { pc = 0 } }
                    }
                    set_temp_variable = { pc = rchange } 
                }
                1 = { 
                    modifier = { 
                        factor = 0 
                        NOT = { check_variable = { mc = 0 } }
                    }
                    set_temp_variable = { mc = rchange } 
                }
                1 = { 
                    modifier = { 
                        factor = 0 
                        NOT = { check_variable = { sc = 0 } }
                    }
                    set_temp_variable = { sc = rchange } 
                }
                1 = { 
                    modifier = { 
                        factor = 0 
                        NOT = { check_variable = { bc = 0 } }
                    }
                    set_temp_variable = { bc = rchange } 
                }
                1 = { 
                    modifier = { 
                        factor = 0 
                        NOT = { check_variable = { ac = 0 } }
                    }
                    set_temp_variable = { ac = rchange } 
                }
            }
        }
        class_pop_satisfactions_change = yes
    }
}

			add_to_variable = { FRA.pop_workers_sore = -0.1 }
		}

            integer = yes

extremists_buff_judge = {
    if = {
        limit = {
            check_variable = { vertes_status = 0 }
            check_variable = { commando_status = 0 }
            check_variable = { mafia_status = 0 }
            check_variable = { bonnot_status = 0 }
        }
        country_event = { id = cofsociety.4 days = 3 }
    }

    set_temp_variable = { t = vertes_status }
    add_to_temp_variable = { t = commando_status }
    add_to_temp_variable = { t = mafia_status }
    add_to_temp_variable = { t = bonnot_status }

    set_variable = { extremist_political_power_cost = t }
    multiply_variable = { extremist_political_power_cost = 0.025 }

    set_variable = { extremist_stability_factor = t }
    multiply_variable = { extremist_stability_factor = -0.005 }

    set_variable = { vertes_buff0 = vertes_status }
    multiply_variable = { vertes_buff0 = 0.1 }
    if = {
        limit = { check_variable = { vertes_attitude = -1 } }
        multiply_variable = { vertes_buff0 = -1 }
    }
    set_variable = { vertes_buff1 = vertes_status }
    multiply_variable = { vertes_buff1 = -0.015 }
    set_variable = { vertes_buff2 = vertes_status }
    multiply_variable = { vertes_buff2 = -0.02 }
    if = {
        limit = { check_variable = { vertes_attitude = 1 } }
        multiply_variable = { vertes_buff1 = 0 }
        multiply_variable = { vertes_buff2 = 0 }
    }

    set_variable = { commando_buff0 = commando_status }
    multiply_variable = { commando_buff0 = -0.02 }
    if = {
        limit = { check_variable = { commando_attitude = -1 } }
        multiply_variable = { commando_buff0 = -1 }
    }
    set_variable = { commando_buff1 = commando_status }
    multiply_variable = { commando_buff1 = -0.01 }
    set_variable = { commando_buff2 = commando_status }
    multiply_variable = { commando_buff2 = -0.01 }
    if = {
        limit = { check_variable = { commando_attitude = 1 } }
        multiply_variable = { commando_buff1 = 0 }
        multiply_variable = { commando_buff2 = 0 }
    }

    set_variable = { mafia_buff0 = mafia_status }
    multiply_variable = { mafia_buff0 = 0.2 }
    if = {
        limit = { check_variable = { mafia_attitude = -1 } }
        multiply_variable = { mafia_buff0 = -1 }
    }
    set_variable = { mafia_buff1 = mafia_status }
    multiply_variable = { mafia_buff1 = 0.03 }
    set_variable = { mafia_buff2 = mafia_status }
    multiply_variable = { mafia_buff2 = -0.01 }
    if = {
        limit = { check_variable = { mafia_attitude = 1 } }
        multiply_variable = { mafia_buff1 = 0 }
        multiply_variable = { mafia_buff2 = 0 }
    }

    set_variable = { bonnot_buff0 = bonnot_status }
    multiply_variable = { bonnot_buff0 = 0.1 }
    if = {
        limit = { check_variable = { bonnot_attitude = -1 } }

    }
