	name = if_bill_pass
	text = {
		trigger = {
			if = {
				limit = { has_country_flag = bill_pass_need_434 }
				check_variable = { agreement_vote_count > 433 }
			}
			else = { check_expr = { value = 650 subtract = abstention_vote_count divide = 2 less_than = agreement_vote_count } }
		}
		localization_key = bill_pass
	}
	text = { localization_key = bill_not_pass }
}
