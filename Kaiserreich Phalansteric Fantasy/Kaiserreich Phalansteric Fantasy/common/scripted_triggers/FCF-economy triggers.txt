	### 作用域：地块；主要作用：判断指定类型的本地工厂数量是否高于目标值。
	state_local_factories_above_target = {
		custom_trigger_tooltip = {
			if = {
				limit = { check_variable = { target_building_type = 0 } }
				set_temp_variable = { selected_factory_count = state_factory }
			}
			else_if = {
				limit = { check_variable = { target_building_type = 1 } }
				set_temp_variable = { selected_factory_count = state_economy_industry }
			}
			else_if = {
				limit = { check_variable = { target_building_type = 2 } }
				set_temp_variable = { selected_factory_count = state_economy_arms }
			}
			else_if = {
				limit = { check_variable = { target_building_type = 3 } }
				set_temp_variable = { selected_factory_count = state_economy_dockyard }
			}
			tooltip = local_factories_more_than_x
			check_variable = { selected_factory_count > target_factory_count }
		}
	}
