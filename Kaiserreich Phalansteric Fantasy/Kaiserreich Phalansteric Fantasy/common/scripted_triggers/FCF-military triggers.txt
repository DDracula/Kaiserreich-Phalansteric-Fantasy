### 军队内部构成 ###
####################
	### 作用域：国家；主要作用：判断军官影响力是否高于士官与士兵。
	military_pop_officers_is_largest = {
		custom_trigger_tooltip = {
			tooltip = {
				localization_key = FCF_society_pop_most_influential_tt
				POP = pop_army_officers
			}
			check_variable = { pop_army_officers > pop_army_ncos }
			check_variable = { pop_army_officers > pop_army_soldiers }
		}
	}
	### 作用域：国家；主要作用：判断士官影响力是否高于军官与士兵。
	military_pop_NCOs_is_largest = {
		custom_trigger_tooltip = {
			tooltip = {
				localization_key = FCF_society_pop_most_influential_tt
				POP = pop_army_ncos
			}
			check_variable = { pop_army_ncos > pop_army_officers }
			check_variable = { pop_army_ncos > pop_army_soldiers }
		}
	}
	### 作用域：国家；主要作用：按当前比较条件判断士兵是否为军队内部影响力最高阶层。
	military_pop_soldiers_is_largest = {
		custom_trigger_tooltip = {
			tooltip = {
				localization_key = FCF_society_pop_most_influential_tt
				POP = pop_army_soldiers
			}
			check_variable = { pop_army_soldiers > pop_army_officers }
			check_variable = { pop_army_soldiers > pop_army_soldiers }
		}
	}
