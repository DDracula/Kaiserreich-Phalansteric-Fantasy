    party_attitude_judge = {
        meta_trigger = {
            text = {
                custom_trigger_tooltip = {
                    if = { 
                        limit = { check_variable = { target_attitude = 5 } }
                        check_variable = { [PARTY]_attitude > 6 }
                    }
                    else_if = { 
                        limit = { check_variable = { target_attitude = 4 } }
                        check_variable = { [PARTY]_attitude < 7 }
                        check_variable = { [PARTY]_attitude > 2 }
                    }
                    else_if = { 
                        limit = { check_variable = { target_attitude = 3 } }
                        check_variable = { [PARTY]_attitude < 3}
                        check_variable = { [PARTY]_attitude > -3 }
                    }
                    else_if = { 	
                        limit = { check_variable = { target_attitude = 2 } }
                        check_variable = { [PARTY]_attitude < -2 }
                        check_variable = { [PARTY]_attitude > -7 }
                    }    
                    else_if = { 	
                        limit = { check_variable = { target_attitude = 1 } }
                        check_variable = { [PARTY]_attitude < -6 }
                    }
                    tooltip = {
                        localization_key = party_attitude_judge_tt
                        PARTY = party_[PARTY]
                    }
                }
            }
            PARTY = "[party_var_name]"
        }
    }
