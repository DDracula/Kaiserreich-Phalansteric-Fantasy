	### 作用域：国家；主要作用：判断目标 CSP 代表组织的态度是否不低于反对档位。
	CSP_attitude_trigger_not_against = {
		custom_trigger_tooltip = {
			if = {
				limit = { check_variable = { target_CSP_group = 1 } }
				set_temp_variable = { target_CSP_attitude_value = CAE_attitude }
			}
			else_if = {
				limit = { check_variable = { target_CSP_group = 2 } }
				set_temp_variable = { target_CSP_attitude_value = CE_attitude }
			}
			else_if = {
				limit = { check_variable = { target_CSP_group = 3 } }
				set_temp_variable = { target_CSP_attitude_value = CSI_attitude }
			}
			else_if = {
				limit = { check_variable = { target_CSP_group = 4 } }
				set_temp_variable = { target_CSP_attitude_value = CC_attitude }
			}
			else_if = {
				limit = { check_variable = { target_CSP_group = 5 } }
				set_temp_variable = { target_CSP_attitude_value = CGFF_attitude }
			}
			tooltip = CSP_attitude_trigger_not_against_tt
			check_variable = { target_CSP_attitude_value > -2 }
		}
	}

	#set_temp_variable = { target_balance_level = 1 }
	#FCF_politics_CGF_CSP_balance_evaluate = yes
	### 作用域：国家；主要作用：按目标档位判断 CGF 与 CSP 的权力平衡是否达到阈值。
	FCF_politics_CGF_CSP_balance_evaluate = {
		custom_trigger_tooltip = {
			if = {
				limit = { check_variable = { target_balance_level = 5 } }
				OR = {
					check_variable = { politics_CGF_CSP_balance > 90 }
					check_variable = { politics_CGF_CSP_balance = 90 }
				}
			}
			else_if = {
				limit = { check_variable = { target_balance_level = 4 } }
				OR = {
					check_variable = { politics_CGF_CSP_balance > 70 }
					check_variable = { FRA.politics_CGF_CSP_balance = 70 }
				}
			}
			else_if = {
				limit = { check_variable = { target_balance_level = 3 } }
				OR = {
					check_variable = { politics_CGF_CSP_balance > 50 }
					check_variable = { FRA.politics_CGF_CSP_balance = 50 }
				}
			}
			else_if = {
				limit = { check_variable = { target_balance_level = 2 } }
				OR = {
					check_variable = { politics_CGF_CSP_balance > 30 }
					check_variable = { FRA.politics_CGF_CSP_balance = 30 }
				}
			}
			else_if = {
				limit = { check_variable = { target_balance_level = 1 } }
				OR = {
					check_variable = { politics_CGF_CSP_balance > 10 }
					check_variable = { FRA.politics_CGF_CSP_balance = 10 }
				}
			}

			else_if = {
				limit = { check_variable = { target_balance_level = -1 } }
				OR = {
					check_variable = { politics_CGF_CSP_balance < -10 }
					check_variable = { politics_CGF_CSP_balance = -10 }
				}
			}
			else_if = {
				limit = { check_variable = { target_balance_level = -2 } }
				OR = {
					check_variable = { politics_CGF_CSP_balance < -30 }
					check_variable = { FRA.politics_CGF_CSP_balance = -30 }
				}
			}
			else_if = {
				limit = { check_variable = { target_balance_level = -3 } }
				OR = {
					check_variable = { politics_CGF_CSP_balance < -50 }
					check_variable = { FRA.politics_CGF_CSP_balance = 50 }
				}
			}
			else_if = {
				limit = { check_variable = { target_balance_level = -4 } }
				OR = {
					check_variable = { politics_CGF_CSP_balance < -70 }
					check_variable = { FRA.politics_CGF_CSP_balance = 70 }
				}
			}
			else_if = {
				limit = { check_variable = { target_balance_level = -5 } }
				OR = {
					check_variable = { politics_CGF_CSP_balance < 90 }
					check_variable = { FRA.politics_CGF_CSP_balance = 90 }
				}
			}
			tooltip = FCF_politics_CGF_CSP_balance_evaluate_tt
		}
	}

	#set_temp_variable = { target_party_id = 1 }
	#set_temp_variable = { target_attitude_level = 1 }
	#politics_party_attitude_evaluate = yes
	### 作用域：国家；主要作用：判断目标党派的态度是否处于指定档位。
	politics_party_attitude_evaluate = {
		meta_trigger = {
			text = {
				custom_trigger_tooltip = {
					if = {
						limit = { check_variable = { target_attitude_level = 5 } }
						check_variable = { [PARTY]_attitude > 6 }
					}
					else_if = {
						limit = { check_variable = { target_attitude_level = 4 } }
						check_variable = { [PARTY]_attitude < 7 }
						check_variable = { [PARTY]_attitude > 2 }
					}
					else_if = {
						limit = { check_variable = { target_attitude_level = 3 } }
						check_variable = { [PARTY]_attitude < 3}
						check_variable = { [PARTY]_attitude > -3 }
					}
					else_if = {
						limit = { check_variable = { target_attitude_level = 2 } }
						check_variable = { [PARTY]_attitude < -2 }
						check_variable = { [PARTY]_attitude > -7 }
					}
					else_if = {
						limit = { check_variable = { target_attitude_level = 1 } }
						check_variable = { [PARTY]_attitude < -6 }
					}
					tooltip = {
						localization_key = politics_party_attitude_evaluate_tt
						PARTY = party_[PARTY]
					}
				}
			}
			PARTY = "[party_var_name]"
		}
	}

	politics_party_dominant_faction_evaluate = {
		meta_trigger = {
			text = {
				custom_trigger_tooltip = {
					check_variable = { [PARTY]_faction^0 = target_faction_id }
					tooltip = {
						localization_key = politics_party_dominant_faction_evaluate_tt
						PARTY = party_[PARTY]
						FACTION = [PARTY]_faction_[FACTION]
					}
				}
			}
			PARTY = "[party_var_name]"
			FACTION = "[?target_faction_id]"
		}
	}
