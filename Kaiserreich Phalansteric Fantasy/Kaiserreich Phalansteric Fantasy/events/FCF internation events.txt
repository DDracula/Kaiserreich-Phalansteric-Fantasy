			multiply_temp_variable = { synd_t = 2 }
			multiply_temp_variable = { radsoc_t = 2 }

		}

		add_popularity = { ideology = syndicalist popularity = synd_t }
		add_popularity = { ideology = radical_socialist popularity = radsoc_t }
		add_popularity = { ideology = social_democrat popularity = socdem_t }
	}
